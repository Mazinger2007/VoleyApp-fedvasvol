# ── Cron examples for FedVas scraper ────────────────────────────────────────
#
# Run every 5 minutes (Linux/macOS crontab):
#
#   */5 * * * * /usr/bin/php /var/www/html/fedbasket/scraper/scraper.php >> /var/log/fedvas-scraper.log 2>&1
#
# Run every 5 minutes, debug mode (verbose output):
#
#   */5 * * * * /usr/bin/php /var/www/html/fedbasket/scraper/scraper.php --debug >> /var/log/fedvas-scraper.log 2>&1
#
# XAMPP on Windows — Task Scheduler example:
#
#   Program:  C:\xampp\php\php.exe
#   Arguments: C:\xampp\htdocs\fedbasket\scraper\scraper.php
#   Schedule:  Every 5 minutes
#
# Manual test run (limit to 3 competitions, verbose):
#
#   php scraper/scraper.php --debug --limit=3
#
